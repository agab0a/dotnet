- Xây dựng chương trình chính với menu để thực hiện các chức năng: Thêm, Hiển thị, Sắp xếp, Thoát.

using System;
using System.Collections.Generic;
using System.Linq;

class Pilot
{
    public string FullName { get; set; }
    public int BirthYear { get; set; }
    public int Experience { get; set; }
    public string LicenseNumber { get; set; }
    public string ContactInfo { get; set; }

    public Pilot()
    {
    }

    public Pilot(string fullName, int birthYear, int experience, string licenseNumber, string contactInfo)
    {
        FullName = fullName;
        BirthYear = birthYear;
        Experience = experience;
        LicenseNumber = licenseNumber;
        ContactInfo = contactInfo;
    }

    public virtual void Input()
    {
        Console.Write("Nhập họ tên: ");
        FullName = Console.ReadLine();
        Console.Write("Nhập năm sinh: ");
        BirthYear = int.Parse(Console.ReadLine());
        Console.Write("Nhập số năm kinh nghiệm: ");
        Experience = int.Parse(Console.ReadLine());
        Console.Write("Nhập số giấy phép: ");
        LicenseNumber = Console.ReadLine();
        Console.Write("Nhập thông tin liên lạc: ");
        ContactInfo = Console.ReadLine();
    }

    public virtual void Output()
    {
        Console.WriteLine($"Họ tên: {FullName}, Năm sinh: {BirthYear}, Kinh nghiệm: {Experience} năm, Số giấy phép: {LicenseNumber}, Liên lạc: {ContactInfo}");
    }
}

class CommercialPilot : Pilot
{
    public string Airline { get; set; }
    public string AircraftType { get; set; }

    public CommercialPilot()
    {
    }

    public CommercialPilot(string fullName, int birthYear, int experience, string licenseNumber, string contactInfo, string airline, string aircraftType)
        : base(fullName, birthYear, experience, licenseNumber, contactInfo)
    {
        Airline = airline;
        AircraftType = aircraftType;
    }

    public override void Input()
    {
        base.Input();
        Console.Write("Nhập hãng hàng không: ");
        Airline = Console.ReadLine();
        Console.Write("Nhập loại máy bay: ");
        AircraftType = Console.ReadLine();
    }

    public override void Output()
    {
        base.Output();
        Console.WriteLine($"Hãng hàng không: {Airline}, Loại máy bay: {AircraftType}");
    }
}

class Program
{
    static List<CommercialPilot> pilots = new List<CommercialPilot>();

    static void Main(string[] args)
    {
        Console.OutputEncoding = System.Text.Encoding.UTF8; // Hỗ trợ tiếng Việt

        while (true)
        {
            Console.WriteLine("\n=== Menu Quản Lý Phi Công ===");
            Console.WriteLine("1. Thêm phi công");
            Console.WriteLine("2. Hiển thị danh sách phi công");
            Console.WriteLine("3. Sắp xếp theo kinh nghiệm giảm dần");
            Console.WriteLine("4. Thoát");
            Console.Write("Chọn chức năng (1-4): ");

            string choice = Console.ReadLine();

            switch (choice)
            {
                case "1":
                    AddPilot();
                    break;
                case "2":
                    DisplayPilots();
                    break;
                case "3":
                    SortPilotsByExperience();
                    break;
                case "4":
                    Console.WriteLine("Thoát chương trình.");
                    return;
                default:
                    Console.WriteLine("Lựa chọn không hợp lệ. Vui lòng thử lại!");
                    break;
            }
        }
    }

    static void AddPilot()
    {
        CommercialPilot pilot = new CommercialPilot();
        pilot.Input();

        // Kiểm tra trùng số giấy phép
        if (pilots.Any(p => p.LicenseNumber == pilot.LicenseNumber))
        {
            Console.WriteLine("Số giấy phép đã tồn tại! Không thể thêm phi công này.");
            return;
        }

        pilots.Add(pilot);
        Console.WriteLine("Thêm phi công thành công!");
    }

    static void DisplayPilots()
    {
        if (pilots.Count == 0)
        {
            Console.WriteLine("Danh sách phi công trống!");
            return;
        }

        Console.WriteLine("\n=== Danh Sách Phi Công ===");
        foreach (var pilot in pilots)
        {
            pilot.Output();
            Console.WriteLine("-------------------------");
        }
    }

    static void SortPilotsByExperience()
    {
        if (pilots.Count == 0)
        {
            Console.WriteLine("Danh sách phi công trống!");
            return;
        }

        pilots = pilots.OrderByDescending(p => p.Experience).ToList();
        Console.WriteLine("Đã sắp xếp danh sách theo kinh nghiệm giảm dần!");
        DisplayPilots();
    }
}

### Hướng dẫn sử dụng trong Visual Studio 2022
1. **Tạo dự án mới**:
   - Mở Visual Studio 2022.
   - Chọn "Create a new project".
   - Tìm và chọn **Console App** (C#), sau đó nhấn "Next".
   - Đặt tên dự án (ví dụ: `PilotManagement`), chọn vị trí lưu, rồi nhấn "Create".

2. **Thay thế mã nguồn**:
   - Mở file `Program.cs` trong Solution Explorer.
   - Xóa nội dung mặc định và dán toàn bộ mã nguồn từ trên vào.

3. **Chạy chương trình**:
   - Nhấn **F5** hoặc nút "Start" để chạy chương trình.
   - Chương trình sẽ hiển thị menu với các chức năng: Thêm phi công, Hiển thị danh sách, Sắp xếp theo kinh nghiệm, Thoát.

4. **Kiểm tra tính năng**:
   - Chọn "1" để thêm phi công (nhập thông tin như họ tên, năm sinh, kinh nghiệm, số giấy phép, thông tin liên lạc, hãng hàng không, loại máy bay). Chương trình sẽ kiểm tra trùng số giấy phép.
   - Chọn "2" để hiển thị danh sách phi công.
   - Chọn "3" để sắp xếp danh sách theo kinh nghiệm giảm dần.
   - Chọn "4" để thoát.

### Giải thích mã
- **Lớp `Pilot`**: Lớp cơ sở với các thuộc tính `FullName`, `BirthYear`, `Experience`, `LicenseNumber`, `ContactInfo`. Có phương thức `Input` để nhập và `Output` để hiển thị thông tin.
- **Lớp `CommercialPilot`**: Kế thừa từ `Pilot`, bổ sung thuộc tính `Airline` và `AircraftType`. Ghi đè phương thức `Input` và `Output` để xử lý thêm các thuộc tính mới.
- **Chương trình chính**:
  - Sử dụng `List<CommercialPilot>` để lưu danh sách phi công.
  - Menu được xây dựng với vòng lặp `while` và `switch-case` để xử lý các lựa chọn.
  - Kiểm tra trùng `LicenseNumber` khi thêm phi công bằng LINQ (`pilots.Any`).
  - Sắp xếp danh sách theo kinh nghiệm giảm dần bằng LINQ (`OrderByDescending`).
