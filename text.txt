1. Định nghĩa lớp Pilot:

using System;

public class Pilot
{
    // Biến thành viên
    public string FullName { get; set; }
    public int BirthYear { get; set; }
    public int Experience { get; set; }
    public string LicenseNumber { get; set; }
    public string ContactInfo { get; set; }

    // Phương thức nhập thông tin
    public virtual void Input()
    {
        Console.Write("Nhập Họ tên: ");
        FullName = Console.ReadLine();
        Console.Write("Nhập Năm sinh: ");
        BirthYear = int.Parse(Console.ReadLine());
        Console.Write("Nhập số năm kinh nghiệm: ");
        Experience = int.Parse(Console.ReadLine());
        Console.Write("Nhập số bằng lái: ");
        LicenseNumber = Console.ReadLine();
        Console.Write("Nhập thông tin liên lạc: ");
        ContactInfo = Console.ReadLine();
    }

    // Phương thức hiển thị thông tin
    public virtual void Output()
    {
        Console.Write($"{FullName}\t{BirthYear}\t{Experience}\t{LicenseNumber}\t{ContactInfo}");
    }
}

2. Định nghĩa lớp CommercialPilot (kế thừa Pilot):

public class CommercialPilot : Pilot
{
    public string Airline { get; set; }
    public string AircraftType { get; set; }

    // Ghi đè phương thức nhập
    public override void Input()
    {
        base.Input(); // Gọi Input của lớp cha
        Console.Write("Nhập hãng hàng không: ");
        Airline = Console.ReadLine();
        Console.Write("Nhập loại máy bay: ");
        AircraftType = Console.ReadLine();
    }

    // Ghi đè phương thức hiển thị
    public override void Output()
    {
        base.Output(); // Gọi Output của lớp cha
        Console.WriteLine($"\t{Airline}\t{AircraftType}");
    }
}

3. Chương trình chính (menu thao tác):

using System;
using System.Collections.Generic;
using System.Linq;

class Program
{
    static List<CommercialPilot> pilots = new List<CommercialPilot>();

    static void Main(string[] args)
    {
        int choice;
        do
        {
            Console.WriteLine("\n====== MENU ======");
            Console.WriteLine("1. Thêm phi công thương mại");
            Console.WriteLine("2. Hiển thị danh sách phi công");
            Console.WriteLine("3. Sắp xếp theo số năm kinh nghiệm (giảm dần)");
            Console.WriteLine("4. Thoát");
            Console.Write("Chọn chức năng: ");
            choice = int.Parse(Console.ReadLine());

            switch (choice)
            {
                case 1:
                    AddPilot();
                    break;
                case 2:
                    DisplayPilots();
                    break;
                case 3:
                    SortPilots();
                    break;
                case 4:
                    Console.WriteLine("Thoát chương trình.");
                    break;
                default:
                    Console.WriteLine("Lựa chọn không hợp lệ. Vui lòng chọn lại!");
                    break;
            }
        } while (choice != 4);
    }

    // Chức năng thêm phi công
    static void AddPilot()
    {
        CommercialPilot pilot = new CommercialPilot();
        pilot.Input();
        pilots.Add(pilot);
        Console.WriteLine("Đã thêm phi công thành công!");
    }

    // Chức năng hiển thị danh sách
    static void DisplayPilots()
    {
        Console.WriteLine("\nDanh sách phi công:");
        Console.WriteLine("Họ tên\tNăm sinh\tKinh nghiệm\tSố bằng lái\tLiên lạc\tHãng\tLoại máy bay");
        foreach (var pilot in pilots)
        {
            pilot.Output();
        }
    }

    // Chức năng sắp xếp theo experience giảm dần
    static void SortPilots()
    {
        pilots = pilots.OrderByDescending(p => p.Experience).ToList();
        Console.WriteLine("Đã sắp xếp danh sách phi công theo kinh nghiệm giảm dần!");
    }
}

Đã tạo lớp Pilot và lớp kế thừa CommercialPilot.

Đã viết Input và Output chuẩn yêu cầu.

Tạo menu console với các chức năng Thêm / Hiển thị / Sắp xếp / Thoát.

Dữ liệu các trường khi hiển thị cách nhau bằng ký tự tab (\t).

Sắp xếp danh sách theo experience giảm dần.
