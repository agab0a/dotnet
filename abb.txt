Dưới đây là hướng dẫn chi tiết để bạn thực hiện bài tập WPF về quản lý thông tin công nhân. Tôi đã thiết kế ứng dụng theo đúng yêu cầu: giao diện chính với các trường nhập, DataGrid, các nút chức năng (Thêm/Sửa/Xóa/Xem danh sách/Kết thúc), validation với thông báo lỗi (sử dụng MessageBox cho đơn giản, hoặc bạn có thể thêm Label lỗi nếu cần nâng cao), style/template cho Button/TextBox/DataGrid, và Window mới cho "Xem danh sách" (lọc lương >=1000).

### Bước 1: Tạo project
- Mở Visual Studio, tạo WPF App (.NET Framework hoặc .NET Core).
- Đặt tên project: `QuanLyCongNhan`.
- Thêm class `CongNhan.cs`.
- Thêm new Window tên `WindowXemDanhSach`.
- Cả hai Window đều set `WindowStartupLocation="CenterScreen"` trong XAML để xuất hiện giữa màn hình.

### CongNhan.cs (Class dữ liệu)
```csharp
using System;

namespace QuanLyCongNhan
{
    public class CongNhan
    {
        public string MaCongNhan { get; set; }
        public string HoTen { get; set; }
        public DateTime? NgaySinh { get; set; }
        public string GioiTinh { get; set; }
        public string DiaChi { get; set; }
        public double Luong { get; set; }
    }
}
```

### MainWindow.xaml.cs (Code behind)
```csharp
using System;
using System.Collections.ObjectModel;
using System.Linq;
using System.Windows;
using System.Windows.Controls;

namespace QuanLyCongNhan
{
    public partial class MainWindow : Window
    {
        public ObservableCollection<CongNhan> ListCongNhan { get; set; } = new ObservableCollection<CongNhan>();
        private CongNhan selectedCongNhan = null;

        public MainWindow()
        {
            InitializeComponent();
            dgCongNhan.ItemsSource = ListCongNhan;
        }

        // Thêm: Validate và thêm vào DataGrid, hiển thị MessageBox xác nhận
        private void btnThem_Click(object sender, RoutedEventArgs e)
        {
            string error = ValidateInput();
            if (!string.IsNullOrEmpty(error))
            {
                MessageBox.Show(error, "Loi nhap lieu");
                return;
            }

            var cn = CreateCongNhanFromInput();

            ListCongNhan.Add(cn);
            MessageBox.Show($"Da them cong nhan:\nMa: {cn.MaCongNhan}\nHo ten: {cn.HoTen}\nNgay sinh: {cn.NgaySinh?.ToString("dd/MM/yyyy")}\nGioi tinh: {cn.GioiTinh}\nDia chi: {cn.DiaChi}\nLuong: {cn.Luong:N0}");
            ClearInput();
        }

        // Sửa: Chọn từ DataGrid, cập nhật
        private void btnSua_Click(object sender, RoutedEventArgs e)
        {
            if (selectedCongNhan == null)
            {
                MessageBox.Show("Vui long chon cong nhan de sua!");
                return;
            }

            string error = ValidateInput();
            if (!string.IsNullOrEmpty(error))
            {
                MessageBox.Show(error, "Loi nhap lieu");
                return;
            }

            UpdateCongNhanFromInput(selectedCongNhan);
            dgCongNhan.Items.Refresh();
            MessageBox.Show("Da sua thanh cong!");
            ClearInput();
            selectedCongNhan = null;
        }

        // Xóa: Dựa trên mã
        private void btnXoa_Click(object sender, RoutedEventArgs e)
        {
            if (string.IsNullOrEmpty(txtMa.Text))
            {
                MessageBox.Show("Vui long nhap ma cong nhan de xoa!");
                return;
            }

            var cnToRemove = ListCongNhan.FirstOrDefault(cn => cn.MaCongNhan == txtMa.Text);
            if (cnToRemove == null)
            {
                MessageBox.Show("Khong tim thay cong nhan voi ma nay!");
                return;
            }

            ListCongNhan.Remove(cnToRemove);
            MessageBox.Show("Da xoa thanh cong!");
            ClearInput();
        }

        // Xem danh sách: Mở Window mới với lương >=1000
        private void btnXemDanhSach_Click(object sender, RoutedEventArgs e)
        {
            WindowXemDanhSach w = new WindowXemDanhSach(ListCongNhan.Where(cn => cn.Luong >= 1000).ToList());
            w.ShowDialog();
        }

        // Kết thúc: Đóng app
        private void btnKetThuc_Click(object sender, RoutedEventArgs e)
        {
            Application.Current.Shutdown();
        }

        // Chọn hàng trong DataGrid: Đổ dữ liệu lên form
        private void dgCongNhan_SelectionChanged(object sender, SelectionChangedEventArgs e)
        {
            selectedCongNhan = dgCongNhan.SelectedItem as CongNhan;
            if (selectedCongNhan != null)
            {
                txtMa.Text = selectedCongNhan.MaCongNhan;
                txtHoTen.Text = selectedCongNhan.HoTen;
                dpNgaySinh.SelectedDate = selectedCongNhan.NgaySinh;
                if (selectedCongNhan.GioiTinh == "Nam") rbNam.IsChecked = true;
                else if (selectedCongNhan.GioiTinh == "Nu") rbNu.IsChecked = true;
                cbDiaChi.SelectedItem = cbDiaChi.Items.Cast<ComboBoxItem>().FirstOrDefault(item => item.Content.ToString() == selectedCongNhan.DiaChi);
                txtLuong.Text = selectedCongNhan.Luong.ToString();
            }
        }

        // Validate input
        private string ValidateInput()
        {
            if (string.IsNullOrEmpty(txtMa.Text)) return "Ma cong nhan khong duoc rong!";
            if (string.IsNullOrEmpty(txtHoTen.Text)) return "Ho ten khong duoc rong!";
            if (dpNgaySinh.SelectedDate == null) return "Vui long chon ngay sinh!";
            if (!rbNam.IsChecked.Value && !rbNu.IsChecked.Value) return "Vui long chon gioi tinh!";
            if (cbDiaChi.SelectedItem == null) return "Vui long chon dia chi!";
            if (string.IsNullOrEmpty(txtLuong.Text) || !double.TryParse(txtLuong.Text, out double luong) || luong < 0) return "Luong phai la so duong!";
            return null;
        }

        // Tạo CongNhan từ input
        private CongNhan CreateCongNhanFromInput()
        {
            return new CongNhan
            {
                MaCongNhan = txtMa.Text,
                HoTen = txtHoTen.Text,
                NgaySinh = dpNgaySinh.SelectedDate,
                GioiTinh = rbNam.IsChecked == true ? "Nam" : "Nu",
                DiaChi = (cbDiaChi.SelectedItem as ComboBoxItem)?.Content.ToString(),
                Luong = double.Parse(txtLuong.Text)
            };
        }

        // Cập nhật CongNhan từ input
        private void UpdateCongNhanFromInput(CongNhan cn)
        {
            cn.MaCongNhan = txtMa.Text;
            cn.HoTen = txtHoTen.Text;
            cn.NgaySinh = dpNgaySinh.SelectedDate;
            cn.GioiTinh = rbNam.IsChecked == true ? "Nam" : "Nu";
            cn.DiaChi = (cbDiaChi.SelectedItem as ComboBoxItem)?.Content.ToString();
            cn.Luong = double.Parse(txtLuong.Text);
        }

        // Xóa input
        private void ClearInput()
        {
            txtMa.Text = "";
            txtHoTen.Text = "";
            dpNgaySinh.SelectedDate = null;
            rbNam.IsChecked = false;
            rbNu.IsChecked = false;
            cbDiaChi.SelectedIndex = -1;
            txtLuong.Text = "";
        }
    }
}
```

### Hướng dẫn chạy và kiểm tra
- Build và run project.
- Thử thêm công nhân: Nhập đầy đủ, nhấn Thêm → MessageBox xác nhận, thêm vào DataGrid.
- Validation: Nếu sai (rỗng, lương không số), hiển thị MessageBox lỗi.
- Sửa: Chọn hàng trong DataGrid → dữ liệu lên form → chỉnh sửa → nhấn Sửa.
- Xóa: Nhập mã → nhấn Xóa.
- Xem danh sách: Mở Window mới với filter lương >=1000.
- Kết thúc: Đóng app.
- Style: Kiểm tra button bo góc vàng chữ xanh, TextBox focus Arial 10pt #ffc, DataGrid header bold center, cột Lương xanh phải nghìn.

Nếu cần thêm (ví dụ: Label lỗi bên cạnh ô thay MessageBox, hoặc unique mã), cho tôi biết để chỉnh!
