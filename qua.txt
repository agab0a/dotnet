Chào bạn, với tư cách là một AI, tôi phải nói thật lòng rằng việc cày cuốc một môn lập trình như .NET (chuyên về WPF và EF Core) chỉ trong một đêm là một thử thách "rất khoai". Tuy nhiên, chiến thuật "tỉa rễ, hái quả" của bạn – tức là chỉ chọn làm những phần cơ bản nhất, cốt lõi nhất để đủ điểm qua môn và bỏ qua các phần phức tạp – là một chiến lược cực kỳ thông minh trong hoàn cảnh này!
Dựa trên đề thi minh họa quản lý học sinh bạn cung cấp, tôi sẽ chuẩn bị cho bạn một "bộ khung" code cơ bản nhất, bám sát đúng 3 mục tiêu bạn bắt buộc phải làm để bạn học thuộc hoặc tham khảo.
### Bước 1: Điều kiện tiên quyết (Kết nối CSDL)
Trước khi làm gì, bạn **bắt buộc** phải biết cách dùng lệnh Scaffold-DbContext trong *Package Manager Console* để máy tự phát sinh code C# từ Database (bạn không cần tự code tay phần này).
Cú pháp thường dùng:
Scaffold-DbContext "Server=Tên_Máy_Của_Bạn;Database=QLHocSinh;Trusted_Connection=True;" Microsoft.EntityFrameworkCore.SqlServer -OutputDir Models

### Bước 3 & 4: Code Xử Lý Hiện Dữ Liệu & Thêm Dữ Liệu (C#)
Phần này sẽ giải quyết yêu cầu truy vấn hiển thị dữ liệu và thêm bản ghi mới rồi hiển thị lại trên DataGrid.
Trong file MainWindow.xaml.cs, bạn code như sau (Giả sử Context của bạn tên là QlHocSinhContext):
```csharp
using System;
using System.Linq;
using System.Windows;
using WpfApp.Models; // Sửa lại đúng tên namespace chứa Models của bạn

namespace WpfApp
{
    public partial class MainWindow : Window
    {
        // Khởi tạo đối tượng kết nối CSDL
        QlHocSinhContext db = new QlHocSinhContext();

        public MainWindow()
        {
            InitializeComponent();
        }

        // HÀM HIỂN THỊ DỮ LIỆU (MỤC TIÊU 2)
        private void Window_Loaded(object sender, RoutedEventArgs e)
        {
            LoadData();
        }

        private void LoadData()
        {
            // Truy vấn lấy dữ liệu, ghép bảng (join) để lấy Tên Lớp, tính toán lấy Tuổi
            var query = from hs in db.HocSinhs
                        join lop in db.Lops on hs.MaLop equals lop.MaLop
                        select new
                        {
                            MaHs = hs.MaHs,
                            HoTen = hs.HoTen,
                            NgaySinh = hs.NgaySinh,
                            GioiTinh = hs.GioiTinh == true ? "Nam" : "Nữ", // Chuyển đổi boolean sang text
                            ConTbls = hs.ConTbls == true ? "Có" : "Không",
                            TenLop = lop.TenLop,
                            Tuoi = DateTime.Now.Year - hs.NgaySinh.Value.Year // Tính toán độ tuổi
                        };

            // Đổ dữ liệu vào DataGrid
            dgvHocSinh.ItemsSource = query.ToList();
        }

        // HÀM THÊM BẢN GHI MỚI (MỤC TIÊU 3 - KHÔNG CẦN BẮT LỖI PHỨC TẠP)
        private void btnThem_Click(object sender, RoutedEventArgs e)
        {
            try
            {
                // Tạo một đối tượng học sinh mới và lấy dữ liệu từ Giao diện
                HocSinh hsMoi = new HocSinh
                {
                    MaHs = txtMaHS.Text,
                    HoTen = txtHoTen.Text,
                    NgaySinh = dpNgaySinh.SelectedDate,
                    GioiTinh = radNam.IsChecked == true,
                    ConTbls = chkConTBLS.IsChecked == true,
                    MaLop = txtMaLop.Text // Bỏ qua ComboBox, nhập thẳng mã lớp (VD: L01)
                };

                // Thêm vào CSDL
                db.HocSinhs.Add(hsMoi);
                db.SaveChanges(); // Lưu thay đổi

                MessageBox.Show("Thêm thành công!");
                LoadData(); // Gọi lại hàm LoadData để hiển thị dữ liệu mới
            }
            catch (Exception ex)
            {
                MessageBox.Show("Có lỗi xảy ra: " + ex.Message);
            }
        }
    }
}

```
