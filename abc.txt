Chào bạn, tôi đã nhận được ảnh mặt sau của tờ đề thi chính thức (Mã đề 90759). Thật may là cấu trúc đề thi này khớp hoàn toàn với những gì bạn mong đợi, chỉ thay đổi tên bảng một chút: CSDL QLDaoTao gồm 2 bảng là SinhVien và Lop.
Dưới đây là lời giải chi tiết, trọn vẹn cho **Câu 2 (7 điểm)**, được viết theo phong cách tối giản, tập trung vào những phần bạn cần làm để tối ưu điểm số trong phòng thi.
### Bước chuẩn bị: Chạy lệnh Scaffold trong Package Manager Console
Trước khi viết code, bạn mở **Tools > NuGet Package Manager > Package Manager Console** và chạy lệnh này để tự động sinh ra các file Model trong thư mục Models:
```powershell
Scaffold-DbContext "Server=.;Database=QLDaoTao;Trusted_Connection=True;TrustServerCertificate=True" Microsoft.EntityFrameworkCore.SqlServer -OutputDir Models

```
### 1. Code Giao diện (Cập nhật MainWindow.xaml)
Để giải quyết trọn vẹn cả **Câu 1** và tích hợp chuẩn bị cho **Câu 2** (bao gồm cả ComboBox lấy mã từ tên lớp ở câu 2c), bạn sử dụng đoạn code XAML dưới đây:
```xml
<Window x:Class="WpfApp.MainWindow"
        xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
        xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
        Title="QUẢN LÝ SINH VIÊN" Height="500" Width="650" Loaded="Window_Loaded">
    
    <Window.Resources>
        <Style x:Key="EllipseBtn" TargetType="Button">
            <Setter Property="Width" Value="90"/>
            <Setter Property="Height" Value="35"/>
            <Setter Property="Margin" Value="10,0"/>
            <Setter Property="Template">
                <Setter.Value>
                    <ControlTemplate TargetType="Button">
                        <Grid>
                            <Ellipse Fill="{TemplateBinding Background}" Stroke="Gray" StrokeThickness="1"/>
                            <ContentPresenter HorizontalAlignment="Center" VerticalAlignment="Center"/>
                        </Grid>
                    </ControlTemplate>
                </Setter.Value>
            </Setter>
        </Style>

        <Style x:Key="HeaderStyle" TargetType="DataGridColumnHeader">
            <Setter Property="Background" Value="LightBlue"/>
            <Setter Property="FontWeight" Value="Bold"/>
            <Setter Property="HorizontalContentAlignment" Value="Center"/>
        </Style>
    </Window.Resources>

    <Grid Margin="20">
        <Grid.RowDefinitions>
            <RowDefinition Height="Auto"/>
            <RowDefinition Height="Auto"/>
            <RowDefinition Height="Auto"/>
            <RowDefinition Height="*""/>
            <RowDefinition Height="Auto"/>
        </Grid.RowDefinitions>

        <TextBlock Text="QUẢN LÝ SINH VIÊN" FontSize="20" HorizontalAlignment="Center" Grid.Row="0" Margin="0,0,0,15"/>

        <Grid Grid.Row="1" Margin="0,0,0,15">
            <Grid.ColumnDefinitions>
                <ColumnDefinition Width="100"/>
                <ColumnDefinition Width="*"/>
            </Grid.ColumnDefinitions>
            <Grid.RowDefinitions>
                <RowDefinition Height="Auto"/>
                <RowDefinition Height="Auto"/>
                <RowDefinition Height="Auto"/>
                <RowDefinition Height="Auto"/>
            </Grid.RowDefinitions>

            <Label Content="Mã sinh viên:" Grid.Row="0" Grid.Column="0" Margin="5"/>
            <TextBox Name="txtMaSV" Grid.Row="0" Grid.Column="1" Margin="5" Width="150" HorizontalAlignment="Left"/>

            <StackPanel Orientation="Horizontal" Grid.Row="2" Grid.Column="1" Margin="5" VerticalAlignment="Center">
                <RadioButton Name="radNam" Content="Nam" IsChecked="True" Margin="0,0,20,0"/>
                <RadioButton Name="radNu" Content="Nữ"/>
            </StackPanel>

            <Label Content="Mã lớp:" Grid.Row="3" Grid.Column="0" Margin="5"/>
            <ComboBox Name="cboLop" Grid.Row="3" Grid.Column="1" Margin="5" Width="200" HorizontalAlignment="Left"
                      DisplayMemberPath="TenLop" SelectedValuePath="MaLop"/>
        </Grid>

        <DataGrid Name="dgvSinhVien" Grid.Row="3" AutoGenerateColumns="False" CanUserAddRows="False" Margin="0,0,0,15" ColumnHeaderStyle="{StaticResource HeaderStyle}">
            <DataGrid.Columns>
                <DataGridTextColumn Header="Mã sinh viên" Binding="{Binding MaSv}" Width="100"/>
                <DataGridTextColumn Header="Họ tên" Binding="{Binding HoTen}" Width="*"/>
                <DataGridTextColumn Header="Giới tính" Binding="{Binding GioiTinh}" Width="80"/>
                <DataGridTextColumn Header="Tên lớp" Binding="{Binding TenLop}" Width="120"/>
            </DataGrid.Columns>
        </DataGrid>

        <StackPanel Orientation="Horizontal" HorizontalAlignment="Center" Grid.Row="4">
            <Button Content="Thêm" Style="{StaticResource EllipseBtn}" Background="Gold" Click="btnThem_Click"/>
            <Button Content="Thống kê" Style="{StaticResource EllipseBtn}" Background="Pink" Click="btnThongKe_Click"/>
        </StackPanel>
    </Grid>
</Window>

```
### 2. Tạo Window phụ cho chức năng Thống kê (Câu 2d)

### 3. Toàn bộ code xử lý C# (File MainWindow.xaml.cs)
Thay thế toàn bộ nội dung file code-behind bằng đoạn mã tối ưu dưới đây để giải quyết toàn bộ các ý **a, b, c, d** của Câu 2. Hãy nhớ sửa lại tên namespace WpfApp thành tên Project của bạn.
```csharp
using System;
using System.Linq;
using System.Windows;
using WpfApp.Models; // Thay bằng tên Project của bạn để kết nối thư mục Models

namespace WpfApp
{
    public partial class MainWindow : Window
    {
        // Khởi tạo đối tượng Context kết nối cơ sở dữ liệu
        QldaoTaoContext db = new QldaoTaoContext();

        public MainWindow()
        {
            InitializeComponent();
        }

        // Sự kiện Form load lên
        private void Window_Loaded(object sender, RoutedEventArgs e)
        {
            LoadComboBoxLop(); // Tải dữ liệu lớp vào ComboBox (Câu 2c)
            LoadData();        // Tải dữ liệu sinh viên lên lưới (Câu 2a)
        }

        // Câu 2c: Đổ danh sách lớp vào ComboBox hiển thị Tên nhưng chọn lấy Mã
        private void LoadComboBoxLop()
        {
            try
            {
                var dsLop = db.Lops.ToList();
                cboLop.ItemsSource = dsLop;
                if (dsLop.Count > 0)
                {
                    cboLop.SelectedIndex = 0; // Chọn mặc định phần tử đầu tiên
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show("Lỗi tải danh sách lớp: " + ex.Message);
            }
        }

        // Câu 2a: Lấy dữ liệu phối hợp giữa 2 bảng hiển thị lên DataGrid
        private void LoadData()
        {
            try
            {
                var query = from sv in db.SinhViens
                            join l in db.Lops on sv.MaLop equals l.MaLop
                            select new
                            {
                                MaSv = sv.MaSv,
                                HoTen = sv.HoTen,
                                GioiTinh = sv.GioiTinh == true ? "Nam" : "Nữ",
                                TenLop = l.TenLop
                            };

                dgvSinhVien.ItemsSource = query.ToList();
            }
            catch (Exception ex)
            {
                MessageBox.Show("Lỗi hiển thị dữ liệu: " + ex.Message);
            }
        }

        // Câu 2b + 2c: Thực hiện thêm bản ghi và kiểm tra dữ liệu đầu vào
        private void btnThem_Click(object sender, RoutedEventArgs e)
        {
            try
            {
                // Kiểm tra người dùng phải nhập đầy đủ các trường dữ liệu
                if (string.IsNullOrWhiteSpace(txtMaSV.Text) || string.IsNullOrWhiteSpace(txtHoTen.Text) || cboLop.SelectedValue == null)
                {
                    MessageBox.Show("Vui lòng nhập đầy đủ tất cả các trường dữ liệu!", "Thông báo", MessageBoxButton.OK, MessageBoxIcon.Warning);
                    return;
                }

                // Kiểm tra trùng khóa chính (Mã sinh viên) tránh lỗi crash ứng dụng
                var checkTrung = db.SinhViens.FirstOrDefault(x => x.MaSv == txtMaSV.Text.Trim());
                if (checkTrung != null)
                {
                    MessageBox.Show("Mã sinh viên này đã tồn tại! Vui lòng nhập mã khác.", "Lỗi dữ liệu", MessageBoxButton.OK, MessageBoxIcon.Error);
                    return;
                }

                // Khởi tạo thực thể mới để chèn vào database
                SinhVien svMoi = new SinhVien
                {
                    MaSv = txtMaSV.Text.Trim(),
                    HoTen = txtHoTen.Text.Trim(),
                    GioiTinh = radNam.IsChecked == true, // Nếu chọn Nam gán true, ngược lại false
                    MaLop = cboLop.SelectedValue.ToString() // Lấy giá trị mã lớp từ ComboBox (Câu 2c)
                };

                db.SinhViens.Add(svMoi);
                db.SaveChanges(); // Lưu thay đổi xuống CSDL

                MessageBox.Show("Thêm sinh viên mới thành công!", "Thành công");
                LoadData(); // Cập nhật lại danh sách trên giao diện sau khi thêm
            }
            catch (Exception ex)
            {
                MessageBox.Show("Có lỗi hệ thống xảy ra: " + ex.Message);
            }
        }

        // Câu 2d: Xử lý nút Thống kê, nhóm dữ liệu và hiển thị sang cửa sổ mới
        private void btnThongKe_Click(object sender, RoutedEventArgs e)
        {
            try
            {
                // Sử dụng GroupJoin hoặc Left Join để đếm số sinh viên Nữ (GioiTinh == false) theo từng lớp
                var queryThongKe = db.Lops.Select(l => new
                {
                    MaLop = l.MaLop,
                    TenLop = l.TenLop,
                    // Đếm số lượng sinh viên thuộc lớp hiện tại và có giới tính là Nữ
                    SoSvnubiet = db.SinhViens.Count(sv => sv.MaLop == l.MaLop && sv.GioiTinh == false)
                }).ToList();

                // Khởi tạo màn hình Window mới và truyền dữ liệu sang
                ThongKeWindow windowPhu = new ThongKeWindow();
                windowPhu.dgvThongKe.ItemsSource = queryThongKe;
                windowPhu.ShowDialog(); // Hiển thị cửa sổ dạng hộp thoại
            }
            catch (Exception ex)
            {
                MessageBox.Show("Lỗi thực hiện thống kê: " + ex.Message);
            }
        }
    }
}

```
